#!/system/bin/env python3 import json import os import signal import socketserver import subprocess import sys from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer from urllib.parse import urlparse, parse_qs MODDIR = os.environ.get("CLASHCTL_HOME", "/data/adb/modules/clashctl") ROOT = os.path.join(MODDIR, "system", "webroot") CLASHCTL = os.path.join(MODDIR, "system", "bin", "clashctl") def json_response(handler, payload, status=200): body = json.dumps(payload).encode("utf-8") handler.send_response(status) handler.send_header("Content-Type", "application/json") handler.send_header("Content-Length", str(len(body))) handler.end_headers() handler.wfile.write(body) def call_clashctl(args): try: proc = subprocess.run([CLASHCTL] + args, capture_output=True, text=True) stdout = (proc.stdout or "").strip() stderr = (proc.stderr or "").strip() return { "ok": proc.returncode == 0, "code": proc.returncode, "stdout": stdout, "stderr": stderr, } except Exception as exc: return {"ok": False, "code": 1, "stdout": "", "stderr": str(exc)} class Handler(BaseHTTPRequestHandler): def log_message(self, format, *args): return def do_GET(self): parsed = urlparse(self.path) path = parsed.path if path == "/api/status": data = call_clashctl(["status"]) json_response(self, {"status": data}) return if path == "/api/ui": json_response(self, {"url": "http://127.0.0.1:9090/"}) return if path == "/api/version": data = call_clashctl(["version"]) json_response(self, {"version": data}) return if path == "/api/subscriptions": result = call_clashctl(["sub", "list"]) json_response(self, {"subscriptions": result}) return if path == "/api/logs": result = call_clashctl(["logs", "50"]) json_response(self, {"logs": result}) return if path == "/api/config": result = call_clashctl(["config", "show"]) json_response(self, {"config": result}) return # static files if path in ("/", ""): path = "/index.html" safe_path = os.path.normpath(path).lstrip("/") abs_path = os.path.join(ROOT, safe_path) if os.path.isfile(abs_path): with open(abs_path, "rb") as fp: data = fp.read() self.send_response(200) if abs_path.endswith(".html"): self.send_header("Content-Type", "text/html; charset=utf-8") elif abs_path.endswith(".js"): self.send_header("Content-Type", "application/javascript; charset=utf-8") elif abs_path.endswith(".css"): self.send_header("Content-Type", "text/css; charset=utf-8") else: self.send_header("Content-Type", "application/octet-stream") self.send_header("Content-Length", str(len(data))) self.end_headers() self.wfile.write(data) return json_response(self, {"error": "Not found"}, 404) def do_POST(self): parsed = urlparse(self.path) path = parsed.path length = int(self.headers.get("Content-Length", "0")) body = self.rfile.read(length).decode("utf-8", errors="ignore") if path == "/api/start": json_response(self, call_clashctl(["start"])) return if path == "/api/stop": json_response(self, call_clashctl(["stop"])) return if path == "/api/subscriptions": try: payload = json.loads(body) if body else {} except Exception: payload = {} url = payload.get("url") or payload.get("URL") if not url: json_response(self, {"error": "Missing url"}, 400) return result = call_clashctl(["sub", "add", "--use", str(url)]) json_response(self, {"result": result}) return if path == "/api/config": try: payload = json.loads(body) if body else {} except Exception: payload = {} key = payload.get("key") value = payload.get("value") if key and value is not None: result = call_clashctl(["config", "set", f"{key}={value}"]) json_response(self, {"result": result}) return json_response(self, {"error": "Missing key/value"}, 400) return json_response(self, {"error": "Unsupported POST endpoint"}, 404) def main(): port = int(os.environ.get("PORT", "9090")) host = os.environ.get("HOST", "127.0.0.1") try: httpd = ThreadingHTTPServer((host, port), Handler) print(f"webroot listening on http://{host}:{port}/") httpd.serve_forever() except KeyboardInterrupt: pass if __name__ == "__main__": main()
